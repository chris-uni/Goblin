// Example program.

let x = 10;
let y = 10;
let z = x + y;

// Instructions required for this:

store @0 10
store @1 20
add %0 @0 @1
store @3 %0

// Definitions

@N  addressable stored value
%X  temporary value
L0  label definition


// Opcodes

-- Arithmetic

add destionation val1 val2
sub destionation val1 val2
mul destionation val1 val2
div destionation val1 val2
mod destionation val1 val2

-- Logical

not destination val
neg destination val
and destination val1 val2
or destionation val1 val2

-- Memory and Variables

store destination val               (only writes into addressable storage, i.e. @)
load destination val                (only loads into temporary storage, having read only from addressable storage i.e. @ into %)

-- Control Flow

jmp destination                     (destination must be a label here, i.e. L0)    
jmpif destination condition

-- Comparision

eq                                     (equal-to)
neq                                    (not-equal-to)
lt                                     (less-than)
lte                                    (less-than-or-equal-to)
gt                                     (greater-than)
gte                                    (greater-than-or-equal-to)
